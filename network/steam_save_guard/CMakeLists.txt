cmake_minimum_required(VERSION 3.20)
project(re2_coop_steam_save_guard LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build with a Windows x64 toolchain.")
endif()
set(REFRAMEWORK_INCLUDE_DIR "" CACHE PATH "Directory containing reframework/API.h")
if(NOT EXISTS "${REFRAMEWORK_INCLUDE_DIR}/reframework/API.h")
    message(FATAL_ERROR "Set REFRAMEWORK_INCLUDE_DIR.")
endif()
add_library(re2_coop_steam_save_guard SHARED steam_save_guard.cpp storage_guard.cpp storage_reads.cpp read_trace.cpp buffer_digest.cpp)
target_link_libraries(re2_coop_steam_save_guard PRIVATE bcrypt)
target_compile_features(re2_coop_steam_save_guard PRIVATE cxx_std_17)
target_include_directories(re2_coop_steam_save_guard SYSTEM PRIVATE "${REFRAMEWORK_INCLUDE_DIR}")
target_compile_definitions(re2_coop_steam_save_guard PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    set_property(TARGET re2_coop_steam_save_guard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(re2_coop_steam_save_guard PRIVATE /W4 /WX)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_storage_guard test_storage_guard.cpp storage_guard.cpp)
    target_compile_features(test_storage_guard PRIVATE cxx_std_17)
    target_compile_definitions(test_storage_guard PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(test_storage_guard PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_write_guard COMMAND test_storage_guard)
    add_executable(test_storage_reads test_storage_reads.cpp storage_guard.cpp storage_reads.cpp read_trace.cpp buffer_digest.cpp)
    target_link_libraries(test_storage_reads PRIVATE bcrypt)
    target_compile_features(test_storage_reads PRIVATE cxx_std_17)
    target_compile_definitions(test_storage_reads PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(test_storage_reads PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_read_observation COMMAND test_storage_reads)
    add_executable(test_read_trace test_read_trace.cpp read_trace.cpp)
    target_compile_features(test_read_trace PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(test_read_trace PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_read_targets COMMAND test_read_trace)
    add_executable(test_buffer_digest test_buffer_digest.cpp buffer_digest.cpp)
    target_compile_features(test_buffer_digest PRIVATE cxx_std_17)
    target_compile_definitions(test_buffer_digest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(test_buffer_digest PRIVATE bcrypt)
    if(MSVC)
        target_compile_options(test_buffer_digest PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_buffer_digests COMMAND test_buffer_digest)
endif()
