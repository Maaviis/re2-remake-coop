cmake_minimum_required(VERSION 3.20)
project(re2_coop_steam_save_guard LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build with a Windows x64 toolchain.")
endif()
set(REFRAMEWORK_INCLUDE_DIR "" CACHE PATH "Directory containing reframework/API.h")
if(NOT EXISTS "${REFRAMEWORK_INCLUDE_DIR}/reframework/API.h")
    message(FATAL_ERROR "Set REFRAMEWORK_INCLUDE_DIR.")
endif()
add_library(re2_coop_steam_save_guard SHARED steam_save_guard.cpp storage_guard.cpp campaign_folder.cpp runtime_save_locks.cpp)
target_compile_features(re2_coop_steam_save_guard PRIVATE cxx_std_17)
target_include_directories(re2_coop_steam_save_guard SYSTEM PRIVATE "${REFRAMEWORK_INCLUDE_DIR}")
target_compile_definitions(re2_coop_steam_save_guard PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(re2_coop_steam_save_guard PRIVATE bcrypt advapi32)
if(MSVC)
    set_property(TARGET re2_coop_steam_save_guard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(re2_coop_steam_save_guard PRIVATE /W4 /WX)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_storage_guard test_storage_guard.cpp storage_guard.cpp campaign_folder.cpp)
    target_link_libraries(test_storage_guard PRIVATE advapi32)
    target_compile_features(test_storage_guard PRIVATE cxx_std_17)
    target_compile_definitions(test_storage_guard PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(test_storage_guard PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_write_guard COMMAND test_storage_guard)
    add_executable(test_guard_start test_guard_start.cpp)
    target_compile_features(test_guard_start PRIVATE cxx_std_17)
    target_include_directories(test_guard_start SYSTEM PRIVATE "${REFRAMEWORK_INCLUDE_DIR}")
    target_compile_definitions(test_guard_start PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_target_properties(test_guard_start PROPERTIES
        OUTPUT_NAME re2 RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/guard-launch-fixture")
    if(MSVC)
        target_compile_options(test_guard_start PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_guard_start
        COMMAND test_guard_start "$<TARGET_FILE:re2_coop_steam_save_guard>")
    add_executable(test_runtime_save_locks test_runtime_save_locks.cpp runtime_save_locks.cpp)
    target_compile_features(test_runtime_save_locks PRIVATE cxx_std_17)
    target_compile_definitions(test_runtime_save_locks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(test_runtime_save_locks PRIVATE advapi32)
    if(MSVC)
        target_compile_options(test_runtime_save_locks PRIVATE /W4 /WX)
    endif()
    add_test(NAME steam_storage_runtime_locks COMMAND test_runtime_save_locks)
endif()
